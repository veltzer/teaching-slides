## Reading from Files

```bash
# Method 1: while read (line by line)
while IFS= read -r line; do
    echo ">> $line"
done < data.txt

# Method 2: mapfile / readarray (entire file into array)
mapfile -t lines < data.txt
echo "File has ${#lines[@]} lines"
echo "First line: ${lines[0]}"
echo "Last line: ${lines[-1]}"

# Method 3: read entire file into a string
content=$(<data.txt)
echo "$content"
```

---

## The `read` Command

```bash
# Read one line from stdin
read -r line
echo "You typed: $line"

# Read with a prompt
read -r -p "Enter your name: " name

# Read with a timeout (seconds)
read -r -t 5 -p "Quick! " answer || echo "Too slow!"

# Read silently (passwords)
read -r -s -p "Password: " password
echo    # add newline after silent read

# Read N characters
read -r -n 1 -p "Press any key..." key
```

## Reading Specific Fields

```bash
# read splits input by IFS
echo "John:Doe:42" | IFS=: read -r first last age
echo "$first"    # John
echo "$last"     # Doe
echo "$age"      # 42

# Read multiple fields from a file
while IFS=, read -r name email role; do
    echo "$name ($role): $email"
done < employees.csv

# Extra fields go into the last variable
echo "a b c d e" | read -r first second rest
echo "$rest"    # c d e
```

## Writing to Files

```bash
# Overwrite
echo "new content" > file.txt

# Append
echo "more content" >> file.txt

# Write multiple lines
cat > file.txt << 'EOF'
line 1
line 2
line 3
EOF

# printf for formatted output
printf "%-20s %5d\n" "Alice" 95 > report.txt
printf "%-20s %5d\n" "Bob" 87 >> report.txt
```

## `printf` vs `echo`

```bash
# echo is simple but inconsistent across systems
echo -n "no newline"    # works in bash, not all sh
echo -e "tab\there"     # works in bash, not all sh

# printf is consistent and more powerful
printf "no newline"
printf "tab\there\n"
printf "%s is %d years old\n" "Alice" 30

# Format specifiers:
printf "%s\n" "string"      # string
printf "%d\n" 42             # integer
printf "%f\n" 3.14           # float
printf "%05d\n" 42           # zero-padded: 00042
printf "%-20s|\n" "left"     # left-aligned
printf "%20s|\n" "right"     # right-aligned
```

## Reading from Processes

```bash
# Command substitution: capture stdout
output=$(ls -la)
today=$(date +%Y-%m-%d)
kernel=$(uname -r)

# Process substitution: treat output as a file
diff <(ls /dir1) <(ls /dir2)

# Read command output line by line
while IFS= read -r line; do
    echo "Process: $line"
done < <(ps aux --no-headers)

# Read into an array
mapfile -t pids < <(pgrep -f "my_process")
```

## Writing to Processes

```bash
# Pipe is the basic mechanism
echo "data" | command

# Process substitution for writing
command > >(tee log.txt)

# Here-string to feed a command
bc <<< "scale=4; 22/7"

# Here-document for multi-line input
mysql -u root << 'SQL'
SELECT * FROM users WHERE active = 1;
SQL

# Feed multiple lines to a program
python3 << 'PYTHON'
import sys
print(f"Python {sys.version}")
PYTHON
```

## `/proc` Filesystem

```bash
# /proc is a virtual filesystem with process info

# Current process ID
echo $$
cat /proc/$$/status | head -5

# Command line of a process
cat /proc/$$/cmdline | tr '\0' ' '

# File descriptors
ls -la /proc/$$/fd/
# 0 -> /dev/pts/1 (stdin)
# 1 -> /dev/pts/1 (stdout)
# 2 -> /dev/pts/1 (stderr)

# Environment
cat /proc/$$/environ | tr '\0' '\n' | head -5
```

## Temporary Files

```bash
# Create a temp file safely
tmpfile=$(mktemp)
echo "data" > "$tmpfile"

# Create a temp directory
tmpdir=$(mktemp -d)
echo "data" > "$tmpdir/file.txt"

# With a template
tmpfile=$(mktemp /tmp/myapp.XXXXXX)
# Creates: /tmp/myapp.a4Bg7k

# ALWAYS clean up
trap 'rm -rf "$tmpdir"' EXIT

# NEVER do this:
tmpfile="/tmp/myapp_temp"    # predictable = security risk
```

## Coprocesses (`bash` 4.0+)

```bash
# Start a coprocess (bidirectional I/O)
coproc BC { bc -l; }

# Write to its stdin
echo "22/7" >&${BC[1]}

# Read from its stdout
read -r result <&${BC[0]}
echo "Result: $result"    # 3.14285714285714285714

# More operations
echo "sqrt(2)" >&${BC[1]}
read -r result <&${BC[0]}
echo "sqrt(2) = $result"

# Close the coprocess
exec {BC[1]}>&-
wait $BC_PID
```
