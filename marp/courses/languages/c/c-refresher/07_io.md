## Standard I/O in C

- C provides standard input/output operations through `<stdio.h>`
- Three standard streams:
    - `stdin`: Standard input (keyboard by default)
    - `stdout`: Standard output (console by default)
    - `stderr`: Standard error (console by default)

---

## Basic Output: printf()

- `printf()` is used for formatted output
- Syntax: `printf("format string", arg1, arg2, ...);`

```c
int age = 30;
float height = 1.75;
printf("Age: %d, Height: %.2f meters\n", age, height);
// Output: Age: 30, Height: 1.75 meters
```

---

## Common Format Specifiers

- `%d`: Integer
- `%f`: Float
- `%c`: Character
- `%s`: String
- `%x`: Hexadecimal
- `%p`: Pointer

```c
printf("Hex: %x, Pointer: %p\n", 255, &age);
```

---

## Basic Input: scanf()

- `scanf()` is used for formatted input
- Syntax: `scanf("format string", &var1, &var2, ...);`

```c
int num;
char str[50];
printf("Enter a number and a word: ");
scanf("%d %s", &num, str);
printf("You entered: %d and %s\n", num, str);
```

---

## Character I/O: getchar() and putchar()

- `getchar()`: Reads a single character from stdin
- `putchar()`: Writes a single character to stdout

```c
int ch;
printf("Enter a character: ");
ch = getchar();
printf("You entered: ");
putchar(ch);
putchar('\n');
```

---

## String I/O: fgets() and puts()

- `gets()`: **removed in C11** — always caused buffer overflows
- `fgets()`: the replacement — reads a line with a size limit
- `puts()`: writes a string to stdout, appending a newline

```c
char str[100];
printf("Enter a string: ");
fgets(str, sizeof(str), stdin);  // safer alternative to gets()
puts("You entered:");
puts(str);
```

## File I/O: Basic Concepts

- File operations involve:
    1. Opening a file
    1. Reading from or writing to the file
    1. Closing the file
- File operations use `FILE` pointers

---

## Opening and Closing Files

- `fopen()`: Opens a file
- `fclose()`: Closes a file

```c
FILE *file = fopen("example.txt", "r");
if (file == NULL) {
    perror("Error opening file");
    return 1;
}
// Perform file operations here
fclose(file);
```

---

## Reading from Files

- `fscanf()`: Formatted input from a file
- `fgets()`: Reads a line from a file

```c
FILE *file = fopen("data.txt", "r");
int num;
char str[100];

fscanf(file, "%d", &num);
fgets(str, sizeof(str), file);

fclose(file);
```

---

## Writing to Files

- `fprintf()`: Formatted output to a file
- `fputs()`: Writes a string to a file

```c
FILE *file = fopen("output.txt", "w");
int num = 42;
char *str = "Hello, File I/O!";

fprintf(file, "Number: %d\n", num);
fputs(str, file);

fclose(file);
```

---

## Error Handling in I/O

- Always check return values of I/O functions
- Use `perror()` or `strerror()` for error messages

```c
if (ferror(file)) {
    perror("File error");
}

if (feof(file)) {
    puts("End of file reached");
}
```

## Binary File I/O

```c
#include <stdio.h>
#include <stdlib.h>

struct Record {
    int id;
    char name[32];
    double salary;
};

int main(void) {
    /* Write binary records */
    struct Record employees[] = {
        {1, "Alice",   75000.0},
        {2, "Bob",     82000.0},
        {3, "Charlie", 69000.0},
    };
    int count = sizeof(employees) / sizeof(employees[0]);

    FILE *f = fopen("records.bin", "wb");
    if (f == NULL) { perror("fopen"); return 1; }
    fwrite(employees, sizeof(struct Record), count, f);
    fclose(f);

    /* Read binary records back */
    struct Record loaded[3];
    f = fopen("records.bin", "rb");
    if (f == NULL) { perror("fopen"); return 1; }
    size_t read = fread(loaded, sizeof(struct Record), 3, f);
    fclose(f);

    for (size_t i = 0; i < read; i++) {
        printf("ID=%d, Name=%s, Salary=%.2f\n",
               loaded[i].id, loaded[i].name, loaded[i].salary);
    }
    return 0;
}
```

## File Positioning: fseek and ftell

```c
#include <stdio.h>

int main(void) {
    FILE *f = fopen("data.bin", "rb");
    if (f == NULL) { perror("fopen"); return 1; }

    /* Get file size */
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    printf("File size: %ld bytes\n", size);

    /* Go back to beginning */
    fseek(f, 0, SEEK_SET);
    /* equivalent: rewind(f); */

    /* Read specific record (e.g., 3rd record of 4 bytes) */
    fseek(f, 2 * sizeof(int), SEEK_SET);
    int value;
    fread(&value, sizeof(int), 1, f);

    fclose(f);
    return 0;
}
```

Seek origins:

| Constant | Meaning |
|----------|---------|
| `SEEK_SET` | Beginning of file |
| `SEEK_CUR` | Current position |
| `SEEK_END` | End of file |

## Line-by-Line File Processing

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int main(int argc, char *argv[]) {
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <filename>\n", argv[0]);
        return 1;
    }

    FILE *f = fopen(argv[1], "r");
    if (f == NULL) {
        perror("fopen");
        return 1;
    }

    char line[1024];
    int line_num = 0;
    int word_count = 0;

    while (fgets(line, sizeof(line), f) != NULL) {
        line_num++;
        /* Count words (simple: split by spaces) */
        char *token = strtok(line, " \t\n");
        while (token != NULL) {
            word_count++;
            token = strtok(NULL, " \t\n");
        }
    }
```

## Printf Width and Precision Formatting

```c
#include <stdio.h>

int main(void) {
    /* Width specifiers */
    printf("[%10d]\n", 42);         /* [        42] */
    printf("[%-10d]\n", 42);        /* [42        ] */
    printf("[%010d]\n", 42);        /* [0000000042] */

    /* Precision for floats */
    printf("[%10.3f]\n", 3.14159);  /* [     3.142] */
    printf("[%.8f]\n", 3.14159);    /* [3.14159000] */

    /* Precision for strings (max chars) */
    printf("[%.5s]\n", "Hello, World!");  /* [Hello] */

    /* Dynamic width with * */
    int width = 15;
    printf("[%*d]\n", width, 42);   /* [             42] */

    /* Table formatting example */
    printf("%-20s %10s %10s\n", "Name", "Price", "Qty");
    printf("%-20s %10.2f %10d\n", "Widget A", 9.99, 100);
    printf("%-20s %10.2f %10d\n", "Gadget B", 24.50, 42);
    printf("%-20s %10.2f %10d\n", "Thingamajig", 1.25, 9999);

    return 0;
}
```

## Temporary Files and Pipes

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    /* Create a temporary file */
    FILE *tmp = tmpfile();
    if (tmp == NULL) { perror("tmpfile"); return 1; }

    fprintf(tmp, "Temporary data: %d\n", 42);

    /* Read it back */
    rewind(tmp);
    char buf[64];
    fgets(buf, sizeof(buf), tmp);
    printf("From tmpfile: %s", buf);
    fclose(tmp);  /* file is automatically deleted */

    /* popen: run a command and read its output */
    FILE *pipe = popen("ls -la", "r");
    if (pipe == NULL) { perror("popen"); return 1; }

    while (fgets(buf, sizeof(buf), pipe) != NULL) {
        printf("  %s", buf);
    }
    pclose(pipe);

    return 0;
}
```

## Redirecting stderr and Logging

```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

void log_message(FILE *log, const char *level, const char *msg) {
    time_t now = time(NULL);
    struct tm *t = localtime(&now);
    char timebuf[32];
    strftime(timebuf, sizeof(timebuf), "%Y-%m-%d %H:%M:%S", t);
    fprintf(log, "[%s] [%s] %s\n", timebuf, level, msg);
    fflush(log);  /* ensure log is written immediately */
}

int main(void) {
    /* Redirect stderr to a log file */
    FILE *logfile = fopen("app.log", "a");
    if (logfile == NULL) { perror("fopen"); return 1; }

    log_message(logfile, "INFO", "Application started");
    log_message(logfile, "DEBUG", "Processing data...");

    /* Also print errors to stderr */
    fprintf(stderr, "Error messages go to stderr\n");

    log_message(logfile, "INFO", "Application finished");
    fclose(logfile);
    return 0;
}
```

## Summary

- `printf()` and `scanf()` for formatted console I/O
- `getchar()`, `putchar()`, `fgets()`, and `puts()` for character and string I/O
- File I/O involves opening, reading/writing, and closing files
- Use `fread`/`fwrite` for binary I/O, `fgets`/`fprintf` for text I/O
- Use `fseek`/`ftell` for random access in files
- Always handle errors and close files when done
- Use `snprintf` for safe string formatting
- Use appropriate functions based on the type of I/O needed
